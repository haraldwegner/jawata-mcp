// A stand-in ACP agent for the engine's ACP client tests (Sprint 28h, S6). It is a single-file Java program, run as
// `java FakeAcpAgent.java`, so it needs nothing but the JDK on any operating system. It speaks just enough ACP over stdio:
//   initialize, session/new, session/load (replays one history chunk), session/prompt, and the permission request it sends to the client.
// What a prompt does depends on its text:
//   REMEMBER:<word>  keeps <word> for the session (a file under $FAKE_ACP_STATE)    -> "noted"
//   RECALL           answers the word kept for the session, or "nothing"
//   SERVERS          answers the mcpServers it was given for the session
//   TOOL / WRITE     makes a tool call (mcp__jawata__inspect / Edit) and asks permission; answers "permission:<option chosen>"
//   CALL:<title>     the same for a tool of that name
//   SPEND            makes five tool calls, each with a usage update that raises the cost
//   STREAM           says "first", waits one and a half seconds, then says "second" (arrival before the turn ends)
//   HANG             sleeps for a minute without answering
//   DIE              exits with status 3 without answering
import java.io.*;
import java.nio.file.*;
import java.util.*;
import java.util.regex.*;

public class FakeAcpAgent {

    static final Map<String, String> servers = new HashMap<>();
    static int counter;
    /** With FAKE_ACP_LOGIN set, a session is refused with "Authentication required" until the client has called authenticate. */
    static final boolean loginRequired = System.getenv("FAKE_ACP_LOGIN") != null;
    static boolean authenticated;
    static final BufferedReader in = new BufferedReader(new InputStreamReader(System.in));

    public static void main(String[] args) throws Exception {
        Path state = Path.of(System.getenv().getOrDefault("FAKE_ACP_STATE", System.getProperty("java.io.tmpdir")));
        for (String line = in.readLine(); line != null; line = in.readLine()) {
            if (line.isBlank()) {
                continue;
            }
            String method = field(line, "method");
            String id = idOf(line);
            switch (method == null ? "" : method) {
                case "initialize" -> reply(id, "{\"protocolVersion\":1,\"agentCapabilities\":{\"loadSession\":true,\"mcpCapabilities\":{\"http\":true}}"
                    + (loginRequired ? ",\"authMethods\":[{\"id\":\"fake_login\",\"name\":\"Fake login\"}]" : "") + "}");
                case "authenticate" -> {
                    authenticated = "fake_login".equals(field(line, "methodId"));
                    reply(id, "{}");
                }
                case "session/new" -> {
                    if (loginRequired && !authenticated) {
                        send("{\"jsonrpc\":\"2.0\",\"id\":" + id + ",\"error\":{\"code\":-32000,\"message\":\"Authentication required\"}}");
                        continue;
                    }
                    String session = "fake-" + (++counter) + "-" + Long.toHexString(System.nanoTime());
                    servers.put(session, between(line, "\"mcpServers\":"));
                    reply(id, "{\"sessionId\":\"" + session + "\"}");
                }
                case "session/load" -> {
                    String session = field(line, "sessionId");
                    servers.put(session, between(line, "\"mcpServers\":"));
                    chunk(session, "HISTORY of the earlier session");
                    reply(id, "{}");
                }
                case "session/prompt" -> prompt(line, id, state);
                default -> { }
            }
        }
        // With FAKE_ACP_LINGER set the agent does not exit when its input closes, as Cursor's agent measured: the client has to end it.
        if (System.getenv("FAKE_ACP_LINGER") != null) {
            Thread.sleep(60_000);
        }
    }

    static void prompt(String line, String id, Path state) throws Exception {
        String session = field(line, "sessionId");
        String text = field(line, "text");
        Path memory = state.resolve(session + ".txt");
        if (text.contains("EDITS:")) {
            // the brain edits N times, each call reported as pending and then completed (an editing kind, as a real brain's own Edit is), then waits
            // 700 ms, drops $FAKE_ACP_STATE/answered.marker and ANSWERS: a gate that ran for a step before the marker ran while the brain worked
            int edits = Integer.parseInt(text.substring(text.indexOf("EDITS:") + "EDITS:".length()).trim().split("\\D")[0]);
            for (int i = 1; i <= edits; i++) {
                send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                    + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"e" + i
                    + "\",\"title\":\"Edit\",\"kind\":\"edit\",\"status\":\"pending\",\"rawInput\":{}}}}");
                send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                    + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"e" + i + "\",\"status\":\"completed\"}}}");
            }
            Thread.sleep(700);
            Files.writeString(state.resolve("answered.marker"), "answered");
            chunk(session, "edited");
        } else if (text.contains("HIREHANG")) {
            // a HIRED seat's prompt starts with the seat's stance, so a hang is asked for by a marker found anywhere in it (S8's live run)
            Thread.sleep(400_000);
        } else if (text.contains("HIREREMEMBER:")) {
            // a HIRED seat's prompt is the seat's stance and then the brief, so these markers are found anywhere in it (D4): the word kept is the
            // one token after the marker, and a later prompt to the SAME brain session (session/load, same id) can ask for it back
            String after = text.substring(text.indexOf("HIREREMEMBER:") + "HIREREMEMBER:".length()).trim();
            Files.writeString(memory, after.split("\\s")[0]);
            chunk(session, "noted");
        } else if (text.contains("HIRERECALL")) {
            chunk(session, Files.exists(memory) ? Files.readString(memory) : "nothing");
        } else if (text.startsWith("REMEMBER:")) {
            Files.writeString(memory, text.substring("REMEMBER:".length()));
            chunk(session, "noted");
        } else if (text.startsWith("RECALL")) {
            chunk(session, Files.exists(memory) ? Files.readString(memory) : "nothing");
        } else if (text.startsWith("SERVERS")) {
            chunk(session, servers.getOrDefault(session, "none"));
        } else if (text.startsWith("TOOL") || text.startsWith("WRITE") || text.startsWith("CALL:")) {
            String title = text.startsWith("CALL:") ? text.substring("CALL:".length())
                : text.startsWith("TOOL") ? "mcp__jawata__inspect" : "Edit";
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"t1\",\"title\":\"" + title
                + "\",\"kind\":\"read\",\"status\":\"pending\",\"rawInput\":{\"a\":1}}}}");
            send("{\"jsonrpc\":\"2.0\",\"id\":900,\"method\":\"session/request_permission\",\"params\":{\"sessionId\":\"" + session
                + "\",\"toolCall\":{\"toolCallId\":\"t1\",\"title\":\"" + title + "\"},\"options\":["
                + "{\"optionId\":\"allow_once\",\"name\":\"Allow\",\"kind\":\"allow_once\"},"
                + "{\"optionId\":\"reject_once\",\"name\":\"Reject\",\"kind\":\"reject_once\"}]}}");
            String answer = in.readLine();
            String chosen = answer == null ? "none" : field(answer, "optionId");
            chunk(session, "permission:" + (chosen == null ? "cancelled" : chosen));
        } else if (text.startsWith("SHELL:")) {
            // the brain's own shell: the title is "Bash" and the permission request carries the command in rawInput, as the Claude adapter sends it
            String command = text.substring("SHELL:".length());
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"s1\",\"title\":\"Bash\",\"kind\":\"execute\",\"status\":\"pending\","
                + "\"rawInput\":{\"command\":\"" + command + "\"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"id\":901,\"method\":\"session/request_permission\",\"params\":{\"sessionId\":\"" + session
                + "\",\"toolCall\":{\"toolCallId\":\"s1\",\"title\":\"Bash\",\"rawInput\":{\"command\":\"" + command + "\"}},\"options\":["
                + "{\"optionId\":\"allow_once\",\"name\":\"Allow\",\"kind\":\"allow_once\"},"
                + "{\"optionId\":\"reject_once\",\"name\":\"Reject\",\"kind\":\"reject_once\"}]}}");
            String answer = in.readLine();
            String chosen = answer == null ? "none" : field(answer, "optionId");
            chunk(session, "permission:" + (chosen == null ? "cancelled" : chosen));
        } else if (text.startsWith("CURSORCALL")) {
            // As Cursor's agent opens a tool call of an MCP server: a placeholder title and no input, named in a later update.
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"c1\",\"title\":\"MCP: tool\",\"kind\":\"other\",\"status\":\"pending\",\"rawInput\":{}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"c1\",\"title\":\"jawata: health_check\","
                + "\"rawInput\":{\"providerIdentifier\":\"jawata\",\"toolName\":\"health_check\",\"args\":{}}}}}");
            chunk(session, "called");
        } else if (text.startsWith("THINK")) {
            // thinking in two chunks and a plan update, then the answer: the three kinds the contract's R9 and R10 rows carry
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"agent_thought_chunk\",\"content\":{\"type\":\"text\",\"text\":\"Checking divisibility \"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"plan\",\"entries\":[{\"content\":\"try 7\",\"priority\":\"medium\",\"status\":\"completed\"},"
                + "{\"content\":\"say the answer\",\"priority\":\"medium\",\"status\":\"pending\"}]}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"agent_thought_chunk\",\"content\":{\"type\":\"text\",\"text\":\"by 7 and 13.\"}}}}");
            chunk(session, "thought");
        } else if (text.startsWith("SPEND")) {
            for (int i = 1; i <= 5; i++) {
                send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                    + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"s" + i
                    + "\",\"title\":\"mcp__jawata__inspect\",\"kind\":\"read\",\"status\":\"pending\",\"rawInput\":{}}}}");
                send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                    + "\",\"update\":{\"sessionUpdate\":\"usage_update\",\"used\":" + (i * 100) + ",\"size\":1000,\"cost\":{\"amount\":"
                    + (i * 0.25) + ",\"currency\":\"USD\"}}}}");
                Thread.sleep(50);
            }
            chunk(session, "spent");
        } else if (text.startsWith("STREAM")) {
            chunk(session, "first");
            Thread.sleep(1500);
            chunk(session, "second");
        } else if (text.startsWith("HANG")) {
            Thread.sleep(60_000);
        } else if (text.startsWith("DIE")) {
            System.exit(3);
        }
        reply(id, "{\"stopReason\":\"end_turn\"}");
    }

    static void chunk(String session, String text) {
        send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
            + "\",\"update\":{\"sessionUpdate\":\"agent_message_chunk\",\"content\":{\"type\":\"text\",\"text\":\"" + text.replace("\\", "\\\\").replace("\"", "\\\"") + "\"}}}}");
    }

    static void reply(String id, String result) {
        send("{\"jsonrpc\":\"2.0\",\"id\":" + id + ",\"result\":" + result + "}");
    }

    static synchronized void send(String message) {
        System.out.println(message);
        System.out.flush();
    }

    static String idOf(String line) {
        Matcher m = Pattern.compile("\"id\"\\s*:\\s*(\\d+)").matcher(line);
        return m.find() ? m.group(1) : "null";
    }

    static String field(String line, String name) {
        Matcher m = Pattern.compile("\"" + name + "\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"").matcher(line);
        return m.find() ? m.group(1).replace("\\\"", "\"").replace("\\\\", "\\") : null;
    }

    /** The JSON value that follows {@code key}, up to the end of the array it opens. */
    static String between(String line, String key) {
        int at = line.indexOf(key);
        if (at < 0) {
            return "none";
        }
        int start = at + key.length();
        int depth = 0;
        for (int i = start; i < line.length(); i++) {
            char c = line.charAt(i);
            if (c == '[') {
                depth++;
            } else if (c == ']' && --depth == 0) {
                return line.substring(start, i + 1);
            }
        }
        return "none";
    }
}
