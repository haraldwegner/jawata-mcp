// A stand-in ACP agent for the engine's ACP client tests (Sprint 28h, S6). It is a single-file Java program, run as
// `java FakeAcpAgent.java`, so it needs nothing but the JDK on any operating system. It speaks just enough ACP over stdio:
//   initialize, session/new, session/load (replays one history chunk), session/prompt, and the permission request it sends to the client.
// What a prompt does depends on its text:
//   REMEMBER:<word>  keeps <word> for the session (a file under $FAKE_ACP_STATE)    -> "noted"
//   RECALL           answers the word kept for the session, or "nothing"
//   SERVERS          answers the mcpServers it was given for the session
//   TOOL / WRITE     makes a tool call (mcp__jawata__inspect / Edit) and asks permission; answers "permission:<option chosen>"
//   CALL:<title>     the same for a tool of that name
//   SPEND            makes five tool calls, each with a usage update that raises the cost
//   STREAM           says "first", waits one and a half seconds, then says "second" (arrival before the turn ends)
//   HANG             sleeps for a minute without answering
//   DIE              exits with status 3 without answering
import java.io.*;
import java.nio.file.*;
import java.util.*;
import java.util.regex.*;

public class FakeAcpAgent {

    static final Map<String, String> servers = new HashMap<>();
    static int counter;
    /** With FAKE_ACP_LOGIN set, a session is refused with "Authentication required" until the client has called authenticate. */
    static final boolean loginRequired = System.getenv("FAKE_ACP_LOGIN") != null;
    static boolean authenticated;
    /**
     * With FAKE_ACP_MODELS set to {@code value=Name,value=Name,...} the agent offers those models in a "model" config option when it opens a session,
     * is on the first, and moves to another when told through session/set_config_option. A prompt WHICHMODEL answers the value it is on (D54 (a)).
     */
    static final String[] offeredModels = System.getenv().getOrDefault("FAKE_ACP_MODELS", "").isBlank() ? new String[0]
        : System.getenv("FAKE_ACP_MODELS").split(",");
    static String modelNow = offeredModels.length == 0 ? "" : offeredModels[0].split("=")[0];
    /**
     * With FAKE_ACP_MODES set to {@code id,id,...} the agent offers those session modes, as Cursor's agent offers agent, plan and ask, is on the first,
     * and moves when told through session/set_mode. A prompt WHICHMODE answers the mode it is on; WITHHELD answers the tools the client asked it to
     * withhold in the session's {@code _meta.claudeCode.options.disallowedTools}, as the Claude adapter reads them, or "none" (28h D57 (j)).
     */
    static final String[] offeredModes = System.getenv().getOrDefault("FAKE_ACP_MODES", "").isBlank() ? new String[0]
        : System.getenv("FAKE_ACP_MODES").split(",");
    static String modeNow = offeredModes.length == 0 ? "" : offeredModes[0];
    static final Map<String, String> withheld = new HashMap<>();
    /** Whether the client asked the brain, in the session's own options, to load no MCP server but the ones the session names (28h Stage 27); a prompt ONLYOURSERVERS answers it. */
    static final Map<String, String> only = new HashMap<>();

    static String modes() {
        StringBuilder available = new StringBuilder();
        for (String mode : offeredModes) {
            available.append(available.length() == 0 ? "" : ",").append("{\"id\":\"").append(mode).append("\",\"name\":\"").append(mode).append("\"}");
        }
        return "{\"currentModeId\":\"" + modeNow + "\",\"availableModes\":[" + available + "]}";
    }

    static String modelOption() {
        StringBuilder options = new StringBuilder();
        for (String offered : offeredModels) {
            String[] pair = offered.split("=", 2);
            options.append(options.length() == 0 ? "" : ",").append("{\"value\":\"").append(pair[0]).append("\",\"name\":\"").append(pair[1]).append("\"}");
        }
        return "{\"id\":\"model\",\"name\":\"Model\",\"category\":\"model\",\"type\":\"select\",\"currentValue\":\"" + modelNow + "\",\"options\":[" + options + "]}";
    }
    // UTF-8 both ways, as a real agent speaks: on Windows the default is the system's code page, and a hire's name "Joe · developer" came
    // back as "Joe ? developer" (CI, November 2026).
    static final BufferedReader in = new BufferedReader(new InputStreamReader(System.in, java.nio.charset.StandardCharsets.UTF_8));
    static final java.io.PrintStream out = new java.io.PrintStream(new java.io.FileOutputStream(java.io.FileDescriptor.out), true,
        java.nio.charset.StandardCharsets.UTF_8);

    public static void main(String[] args) throws Exception {
        Path state = Path.of(System.getenv().getOrDefault("FAKE_ACP_STATE", System.getProperty("java.io.tmpdir")));
        for (String line = in.readLine(); line != null; line = in.readLine()) {
            if (line.isBlank()) {
                continue;
            }
            String method = field(line, "method");
            String id = idOf(line);
            switch (method == null ? "" : method) {
                case "initialize" -> reply(id, "{\"protocolVersion\":1,\"agentCapabilities\":{\"loadSession\":true,\"mcpCapabilities\":{\"http\":true},\"promptCapabilities\":{\"image\":"
                    + (System.getenv("FAKE_ACP_IMAGES") != null) + "}}"
                    + (loginRequired ? ",\"authMethods\":[{\"id\":\"fake_login\",\"name\":\"Fake login\"}]" : "") + "}");
                case "authenticate" -> {
                    authenticated = "fake_login".equals(field(line, "methodId"));
                    reply(id, "{}");
                }
                case "session/new" -> {
                    if (loginRequired && !authenticated) {
                        send("{\"jsonrpc\":\"2.0\",\"id\":" + id + ",\"error\":{\"code\":-32000,\"message\":\"Authentication required\"}}");
                        continue;
                    }
                    String session = "fake-" + (++counter) + "-" + Long.toHexString(System.nanoTime());
                    servers.put(session, between(line, "\"mcpServers\":"));
                    withheld.put(session, between(line, "\"disallowedTools\":"));
                    only.put(session, line.contains("\"strictMcpConfig\":true") ? "only the named servers" : "every server it can find");
                    reply(id, "{\"sessionId\":\"" + session + "\""
                        + (offeredModels.length == 0 ? "" : ",\"configOptions\":[" + modelOption() + "]")
                        + (offeredModes.length == 0 ? "" : ",\"modes\":" + modes()) + "}");
                }
                case "session/set_mode" -> {
                    String wanted = field(line, "modeId");
                    if (java.util.Arrays.asList(offeredModes).contains(wanted)) {
                        modeNow = wanted;
                        reply(id, "{}");
                    } else {
                        send("{\"jsonrpc\":\"2.0\",\"id\":" + id + ",\"error\":{\"code\":-32602,\"message\":\"no such mode\"}}");
                    }
                }
                case "session/set_config_option" -> {
                    String wanted = field(line, "value");
                    boolean offered = "model".equals(field(line, "configId")) && java.util.Arrays.stream(offeredModels).anyMatch(m -> m.split("=")[0].equals(wanted));
                    if (offered) {
                        modelNow = wanted;
                        reply(id, "{\"configOptions\":[" + modelOption() + "]}");
                    } else {
                        send("{\"jsonrpc\":\"2.0\",\"id\":" + id + ",\"error\":{\"code\":-32602,\"message\":\"no such model\"}}");
                    }
                }
                case "session/load" -> {
                    String session = field(line, "sessionId");
                    servers.put(session, between(line, "\"mcpServers\":"));
                    withheld.put(session, between(line, "\"disallowedTools\":"));
                    only.put(session, line.contains("\"strictMcpConfig\":true") ? "only the named servers" : "every server it can find");
                    chunk(session, "HISTORY of the earlier session");
                    reply(id, offeredModes.length == 0 ? "{}" : "{\"modes\":" + modes() + "}");
                }
                case "session/prompt" -> prompt(line, id, state);
                default -> { }
            }
        }
        // With FAKE_ACP_LINGER set the agent does not exit when its input closes, as Cursor's agent measured: the client has to end it.
        if (System.getenv("FAKE_ACP_LINGER") != null) {
            Thread.sleep(60_000);
        }
    }

    static void prompt(String line, String id, Path state) throws Exception {
        String session = field(line, "sessionId");
        String text = field(line, "text");
        Path memory = state.resolve(session + ".txt");
        if (text.contains("EDITS:")) {
            // the brain edits N times, each call reported as pending and then completed (an editing kind, as a real brain's own Edit is), then waits
            // 700 ms, drops $FAKE_ACP_STATE/answered.marker and ANSWERS: a gate that ran for a step before the marker ran while the brain worked
            int edits = Integer.parseInt(text.substring(text.indexOf("EDITS:") + "EDITS:".length()).trim().split("\\D")[0]);
            for (int i = 1; i <= edits; i++) {
                send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                    + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"e" + i
                    + "\",\"title\":\"Edit\",\"kind\":\"edit\",\"status\":\"pending\",\"rawInput\":{}}}}");
                send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                    + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"e" + i + "\",\"status\":\"completed\"}}}");
            }
            Thread.sleep(700);
            Files.writeString(state.resolve("answered.marker"), "answered");
            chunk(session, "edited");
        } else if (text.contains("HIREHANG")) {
            // a HIRED seat's prompt starts with the seat's stance, so a hang is asked for by a marker found anywhere in it (S8's live run)
            Thread.sleep(400_000);
        } else if (text.contains("HIREREMEMBER:")) {
            // a HIRED seat's prompt is the seat's stance and then the brief, so these markers are found anywhere in it (D4): the word kept is the
            // one token after the marker, and a later prompt to the SAME brain session (session/load, same id) can ask for it back
            String after = text.substring(text.indexOf("HIREREMEMBER:") + "HIREREMEMBER:".length()).trim();
            Files.writeString(memory, after.split("\\s")[0]);
            chunk(session, "noted");
        } else if (text.contains("HIRERECALL")) {
            chunk(session, Files.exists(memory) ? Files.readString(memory) : "nothing");
        } else if (text.contains("ASKTRANSCRIPT")) {
            chunk(session, text.contains("EARLIERWORD") ? "knows" : "does not know");
        } else if (text.contains("WHICHMODE") && !text.contains("WHICHMODEL")) {
            chunk(session, modeNow.isEmpty() ? "no modes" : modeNow);
        } else if (text.contains("ONLYOURSERVERS")) {
            chunk(session, only.getOrDefault(session, "unknown"));
        } else if (text.contains("WITHHELD")) {
            chunk(session, withheld.getOrDefault(session, "none"));
        } else if (text.contains("WHICHMODEL")) {
            chunk(session, modelNow);
        } else if (text.contains("WHICHFOLDER")) {
            chunk(session, java.nio.file.Path.of("").toAbsolutePath().toString());
        } else if (text.contains("ASKGREETING")) {
            chunk(session, text.contains("answered at once") ? "told" : "not told");
        } else if (text.contains("WHICHPROCESS")) {
            chunk(session, String.valueOf(ProcessHandle.current().pid()));
        } else if (text.contains("ASKSTANCE")) {
            chunk(session, text.contains("THE-MAIN-STANCE") ? "with stance" : "no stance");
        } else if (text.contains("COUNTIMAGES")) {
            chunk(session, String.valueOf(line.split("\"type\":\"image\"", -1).length - 1));
        } else if (text.startsWith("REMEMBER:")) {
            Files.writeString(memory, text.substring("REMEMBER:".length()));
            chunk(session, "noted");
        } else if (text.startsWith("RECALL")) {
            chunk(session, Files.exists(memory) ? Files.readString(memory) : "nothing");
        } else if (text.startsWith("SERVERS")) {
            chunk(session, servers.getOrDefault(session, "none"));
        } else if (text.startsWith("TOOL") || text.startsWith("WRITE") || text.startsWith("CALL:")) {
            String title = text.startsWith("CALL:") ? text.substring("CALL:".length())
                : text.startsWith("TOOL") ? "mcp__jawata__inspect" : "Edit";
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"t1\",\"title\":\"" + title
                + "\",\"kind\":\"read\",\"status\":\"pending\",\"rawInput\":{\"a\":1}}}}");
            send("{\"jsonrpc\":\"2.0\",\"id\":900,\"method\":\"session/request_permission\",\"params\":{\"sessionId\":\"" + session
                + "\",\"toolCall\":{\"toolCallId\":\"t1\",\"title\":\"" + title + "\"},\"options\":["
                + "{\"optionId\":\"allow_once\",\"name\":\"Allow\",\"kind\":\"allow_once\"},"
                + "{\"optionId\":\"reject_once\",\"name\":\"Reject\",\"kind\":\"reject_once\"}]}}");
            String answer = in.readLine();
            String chosen = answer == null ? "none" : field(answer, "optionId");
            chunk(session, "permission:" + (chosen == null ? "cancelled" : chosen));
        } else if (text.startsWith("SHELL:")) {
            // the brain's own shell: the title is "Bash" and the permission request carries the command in rawInput, as the Claude adapter sends it
            String command = text.substring("SHELL:".length());
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"s1\",\"title\":\"Bash\",\"kind\":\"execute\",\"status\":\"pending\","
                + "\"rawInput\":{\"command\":\"" + command + "\"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"id\":901,\"method\":\"session/request_permission\",\"params\":{\"sessionId\":\"" + session
                + "\",\"toolCall\":{\"toolCallId\":\"s1\",\"title\":\"Bash\",\"rawInput\":{\"command\":\"" + command + "\"}},\"options\":["
                + "{\"optionId\":\"allow_once\",\"name\":\"Allow\",\"kind\":\"allow_once\"},"
                + "{\"optionId\":\"reject_once\",\"name\":\"Reject\",\"kind\":\"reject_once\"}]}}");
            String answer = in.readLine();
            String chosen = answer == null ? "none" : field(answer, "optionId");
            chunk(session, "permission:" + (chosen == null ? "cancelled" : chosen));
        } else if (text.startsWith("TIMEDCALL")) {
            // One tool call that takes 300 ms between its opening and its completion, then 100 ms of the brain's own work before it answers
            // (28h Stage 19, the hire-time split): the client must read the 300 ms as time in a tool.
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"m1\",\"title\":\"mcp__jawata__inspect\",\"kind\":\"read\",\"status\":\"pending\",\"rawInput\":{}}}}");
            Thread.sleep(300);
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"m1\",\"status\":\"completed\"}}}");
            Thread.sleep(100);
            chunk(session, "timed");
        } else if (text.startsWith("CURSORCALL")) {
            // As Cursor's agent opens a tool call of an MCP server: a placeholder title and no input, named in a later update.
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"c1\",\"title\":\"MCP: tool\",\"kind\":\"other\",\"status\":\"pending\",\"rawInput\":{}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"c1\",\"title\":\"jawata: health_check\","
                + "\"rawInput\":{\"providerIdentifier\":\"jawata\",\"toolName\":\"health_check\",\"args\":{}}}}}");
            chunk(session, "called");
        } else if (text.startsWith("STARTHIRE:")) {
            // A brain's hire call that has just answered: the hire is running, under the plan id given (28h Stage 27, finding 28h-C27-1).
            String plan = text.substring("STARTHIRE:".length()).strip();
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"h1\",\"status\":\"completed\",\"rawOutput\":"
                + "{\"success\":true,\"data\":{\"hireId\":\"" + plan + "\",\"name\":\"Zoe\",\"state\":\"running\"}}}}}");
            chunk(session, "started");
        } else if (text.startsWith("CLAUDECALL")) {
            // As Claude's ACP adapter sends a call (recorded 2026-11-17, 28h D57 (d)): opened with an empty input, the input streamed in later
            // updates, partial and then whole, then updates with no input at all as the call runs and completes.
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"e1\",\"title\":\"Edit\",\"kind\":\"edit\",\"status\":\"pending\",\"rawInput\":{}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"e1\",\"rawInput\":{\"file_path\":\"/p/notes.md\"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"e1\",\"rawInput\":{\"file_path\":\"/p/notes.md\","
                + "\"old_string\":\"colour: red\",\"new_string\":\"colour: blue\"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"e1\",\"rawInput\":null}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"e1\",\"status\":\"completed\"}}}");
            chunk(session, "edited");
        } else if (text.startsWith("OWNEDIT:")) {
            // an edit of the brain's own editor on the file named after the colon (28h Stage 27 (z2)): opened with no input, then the path
            String file = text.substring("OWNEDIT:".length()).trim();
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"g1\",\"title\":\"Edit\",\"kind\":\"edit\",\"status\":\"pending\",\"rawInput\":{}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"g1\",\"rawInput\":{\"file_path\":\"" + file + "\"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"g1\",\"status\":\"completed\"}}}");
            chunk(session, "edited");
        } else if (text.startsWith("CLAUDEWRITE:")) {
            // As Claude's ACP adapter sends the write of a NEW file (recorded in 28i phase A, met live in 28h Stage 27 (z5)): opened as
            // "Preparing file…" with the editor named only in _meta, the path in a later update, and no old text in the diff.
            String file = text.substring("CLAUDEWRITE:".length()).trim();
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"_meta\":{\"claudeCode\":{\"toolName\":\"Write\"}},\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"w1\","
                + "\"title\":\"Preparing file\\u2026\",\"kind\":\"edit\",\"status\":\"pending\",\"rawInput\":{}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"_meta\":{\"claudeCode\":{\"toolName\":\"Write\"}},\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"w1\","
                + "\"title\":\"Write\",\"rawInput\":{\"file_path\":\"" + file + "\"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"_meta\":{\"claudeCode\":{\"toolName\":\"Write\"}},\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"w1\","
                + "\"rawInput\":{\"file_path\":\"" + file + "\",\"content\":\"hi\\n\"},\"content\":[{\"type\":\"diff\",\"path\":\"" + file
                + "\",\"oldText\":null,\"newText\":\"hi\\n\"}]}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"w1\",\"status\":\"completed\"}}}");
            chunk(session, "written");
        } else if (text.startsWith("CURSOREDIT")) {
            // As Cursor's agent sends an edit of its own (measured 2026-11-17, 28h D57 (d)): opened with no input, then only the path,
            // and the replacement itself only as the diff of the completed call.
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"f1\",\"title\":\"Edit File\",\"kind\":\"edit\",\"status\":\"pending\",\"rawInput\":{}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"f1\",\"title\":\"Edit `/p/notes.md`\","
                + "\"rawInput\":{\"path\":\"/p/notes.md\"},\"locations\":[{\"path\":\"/p/notes.md\"}]}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"f1\",\"status\":\"in_progress\"}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call_update\",\"toolCallId\":\"f1\",\"status\":\"completed\",\"content\":[{\"type\":\"diff\","
                + "\"path\":\"/p/notes.md\",\"oldText\":\"colour: red\\n\",\"newText\":\"colour: blue\\n\"}]}}}");
            chunk(session, "edited");
        } else if (text.startsWith("THINK")) {
            // thinking in two chunks and a plan update, then the answer: the three kinds the contract's R9 and R10 rows carry
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"agent_thought_chunk\",\"content\":{\"type\":\"text\",\"text\":\"Checking divisibility \"}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"plan\",\"entries\":[{\"content\":\"try 7\",\"priority\":\"medium\",\"status\":\"completed\"},"
                + "{\"content\":\"say the answer\",\"priority\":\"medium\",\"status\":\"pending\"}]}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"agent_thought_chunk\",\"content\":{\"type\":\"text\",\"text\":\"by 7 and 13.\"}}}}");
            chunk(session, "thought");
        } else if (text.startsWith("SPEND")) {
            spend(session);
            chunk(session, "spent");
        } else if (text.contains("A hire you started has ended")) {
            // a report from a hire this brain started (Stage 24): it says which hire, and a report whose answer is SPEND makes the five tool calls again
            chunk(session, "report read: " + text.substring(text.indexOf("Hire: ")).split("\n")[0]);
            if (text.contains("Answer: SPEND")) {
                spend(session);
            }
        } else if (text.startsWith("STREAM")) {
            chunk(session, "first");
            Thread.sleep(1500);
            chunk(session, "second");
        } else if (text.startsWith("WAITCANCEL")) {
            // says it is waiting, then reads until the client cancels the turn, which is how a real brain ends one
            chunk(session, "waiting");
            for (String next = in.readLine(); next != null && !"session/cancel".equals(field(next, "method")); next = in.readLine()) {
            }
            reply(id, "{\"stopReason\":\"cancelled\"}");
            return;
        } else if (text.startsWith("HANG")) {
            Thread.sleep(60_000);
        } else if (text.startsWith("DIE")) {
            System.exit(3);
        }
        reply(id, "{\"stopReason\":\"end_turn\"}");
    }

    /** Five tool calls, each with a usage update that raises the cost. */
    static void spend(String session) throws Exception {
        for (int i = 1; i <= 5; i++) {
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"tool_call\",\"toolCallId\":\"s" + i
                + "\",\"title\":\"mcp__jawata__inspect\",\"kind\":\"read\",\"status\":\"pending\",\"rawInput\":{}}}}");
            send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
                + "\",\"update\":{\"sessionUpdate\":\"usage_update\",\"used\":" + (i * 100) + ",\"size\":1000,\"cost\":{\"amount\":"
                + (i * 0.25) + ",\"currency\":\"USD\"}}}}");
            Thread.sleep(50);
        }
    }

    static void chunk(String session, String text) {
        send("{\"jsonrpc\":\"2.0\",\"method\":\"session/update\",\"params\":{\"sessionId\":\"" + session
            + "\",\"update\":{\"sessionUpdate\":\"agent_message_chunk\",\"content\":{\"type\":\"text\",\"text\":\"" + text.replace("\\", "\\\\").replace("\"", "\\\"") + "\"}}}}");
    }

    static void reply(String id, String result) {
        send("{\"jsonrpc\":\"2.0\",\"id\":" + id + ",\"result\":" + result + "}");
    }

    static synchronized void send(String message) {
        out.println(message);
        out.flush();
    }

    static String idOf(String line) {
        Matcher m = Pattern.compile("\"id\"\\s*:\\s*(\\d+)").matcher(line);
        return m.find() ? m.group(1) : "null";
    }

    static String field(String line, String name) {
        Matcher m = Pattern.compile("\"" + name + "\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"").matcher(line);
        return m.find() ? m.group(1).replace("\\\"", "\"").replace("\\\\", "\\") : null;
    }

    /** The JSON value that follows {@code key}, up to the end of the array it opens. */
    static String between(String line, String key) {
        int at = line.indexOf(key);
        if (at < 0) {
            return "none";
        }
        int start = at + key.length();
        int depth = 0;
        for (int i = start; i < line.length(); i++) {
            char c = line.charAt(i);
            if (c == '[') {
                depth++;
            } else if (c == ']' && --depth == 0) {
                return line.substring(start, i + 1);
            }
        }
        return "none";
    }
